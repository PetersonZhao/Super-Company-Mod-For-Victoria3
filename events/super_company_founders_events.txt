# Run one day after establishment, after vanilla has finished creating the default executive.
# The saved company scope travels with this event, so simultaneous foundations remain independent.
super_company_founders.1 = {
	type = country_event
	hidden = yes

	immediate = {
		# A destroyed or transferred company must not leave behind an unattached founder.
		# Keep this guard in immediate: even an explicitly triggered event must recheck its target.
		if = {
			limit = {
				exists = scope:super_company_founding_company
				any_company = {
					this = scope:super_company_founding_company
				}
			}
			religion = { save_scope_as = super_company_founder_religion }
			super_company_founder_appoint = {
				COMPANY = company_global_automotive_industries
				TEMPLATE = super_company_carl_benz
				CLAIMED = super_company_founder_benz_claimed
