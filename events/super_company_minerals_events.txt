# The yearly proposal. Expiry always chooses postponement.
super_company_minerals.1 = {
	type = country_event
	placement = root
	title = super_company_minerals.1.t
	desc = super_company_minerals.1.d
	flavor = super_company_minerals.1.f
	event_image = { video = "ip4_mining_scene" }
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		super_company_minerals_can_start = yes
	}
	immediate = {
		super_company_minerals_begin = yes
	}

	option = {
		name = super_company_minerals.1.a
		trigger = {
			super_company_minerals_country_eligible = yes
			any_scope_state = { super_company_minerals_state_eligible = yes }
		}
		custom_tooltip = super_company_minerals_choose_tt
		hidden_effect = {
			super_company_minerals_build_state_list = yes
			trigger_event = { id = super_company_minerals.2 }
		}
	}
	option = {
		name = super_company_minerals.cancel
		default_option = yes
		custom_tooltip = super_company_minerals_cancel_tt
		hidden_effect = { super_company_minerals_finish = yes }
	}
}

# Only event-local scopes name the buttons. Clear and repopulate them on every page.
super_company_minerals.2 = {
	type = country_event
	is_popup = yes
	placement = root
	title = super_company_minerals.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:super_company_minerals_state_1 }
				desc = super_company_minerals.2.d
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = super_company_minerals.2.empty.d
			}
		}
	}
	flavor = super_company_minerals.2.f
	event_image = { video = "ip4_mining_scene" }
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	# Keep cancellation available even if the company is dissolved while choosing.
	trigger = { has_variable = super_company_minerals_pending }
	immediate = { super_company_minerals_prepare_page = yes }

	option = {
		name = super_company_minerals.2.state_1
		trigger = {
			super_company_minerals_country_eligible = yes
			exists = scope:super_company_minerals_state_1
			scope:super_company_minerals_state_1 = {
				owner = root

					super_company_minerals_can_confirm = yes

		name = super_company_minerals.change_resource
