	event_image = { video = "southamerica_factory_opening" }
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		super_company_oil_can_start = yes
	}
	immediate = {
		super_company_oil_begin = yes
	}

	option = {
		name = super_company_oil.1.a
		trigger = {
			super_company_oil_country_eligible = yes
			any_scope_state = { super_company_oil_state_eligible = yes }
		}
		custom_tooltip = super_company_oil_choose_tt
		hidden_effect = {
			super_company_oil_build_state_list = yes
			trigger_event = { id = super_company_oil.2 }
		}
	}
	option = {
		name = super_company_oil.cancel
		default_option = yes
		custom_tooltip = super_company_oil_cancel_tt
		hidden_effect = { super_company_oil_finish = yes }
	}
}

# Only event-local scopes name the buttons. Clear and repopulate them on every page.
super_company_oil.2 = {
	type = country_event
	placement = root
	title = super_company_oil.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:super_company_oil_state_1 }
				desc = super_company_oil.2.d
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = super_company_oil.2.empty.d
			}
		}

	flavor = super_company_oil.2.f
	event_image = { video = "southamerica_factory_opening" }
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	# Keep cancellation available even if the company is dissolved while choosing.
	trigger = { has_variable = super_company_oil_pending }
	immediate = { super_company_oil_prepare_page = yes }

	option = {
		name = super_company_oil.2.state_1
		trigger = {
			super_company_oil_country_eligible = yes
