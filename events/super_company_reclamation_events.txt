namespace = super_company_reclamation

# The yearly proposal. Expiry always chooses postponement.
super_company_reclamation.1 = {
	type = country_event
	placement = root
	title = super_company_reclamation.1.t
	desc = super_company_reclamation.1.d
	flavor = super_company_reclamation.1.f
	event_image = { video = "europenorthamerica_russian_serfs" }
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		super_company_reclamation_can_start = yes
	}
	immediate = {
		super_company_reclamation_begin = yes
	}

	option = {
		name = super_company_reclamation.1.a
		trigger = {
			super_company_reclamation_country_eligible = yes
			any_scope_state = { super_company_reclamation_state_eligible = yes }
		}
		custom_tooltip = super_company_reclamation_choose_tt
		hidden_effect = {
			super_company_reclamation_build_state_list = yes
			trigger_event = { id = super_company_reclamation.2 }
		}
	}
	option = {
		name = super_company_reclamation.cancel
		default_option = yes
		custom_tooltip = super_company_reclamation_cancel_tt
		hidden_effect = { super_company_reclamation_finish = yes }
	}
}

# Only event-local scopes name the buttons. Clear and repopulate them on every page.
super_company_reclamation.2 = {
	type = country_event
	is_popup = yes
	placement = root
	title = super_company_reclamation.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:super_company_reclamation_state_1 }
				desc = super_company_reclamation.2.d
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = super_company_reclamation.2.empty.d
			}
		}
	}
	flavor = super_company_reclamation.2.f
	event_image = { video = "europenorthamerica_russian_serfs" }
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	# Keep cancellation available even if the company is dissolved while choosing.
	trigger = { has_variable = super_company_reclamation_pending }
	immediate = { super_company_reclamation_prepare_page = yes }

	option = {
		name = super_company_reclamation.2.state_1
		trigger = {
			super_company_reclamation_country_eligible = yes
			exists = scope:super_company_reclamation_state_1
			scope:super_company_reclamation_state_1 = {
				owner = root
				super_company_reclamation_state_eligible = yes
			}
		}
		custom_tooltip = super_company_reclamation_preview_tt
		hidden_effect = {
			scope:super_company_reclamation_state_1 = {
				save_scope_as = super_company_reclamation_selected_state
			}
			trigger_event = { id = super_company_reclamation.3 }
		}
	}
	option = {
		name = super_company_reclamation.2.state_2
		trigger = {
			super_company_reclamation_country_eligible = yes

				save_scope_as = super_company_reclamation_selected_state
			}
			trigger_event = { id = super_company_reclamation.3 }
		}
	}
	option = {
		name = super_company_reclamation.2.state_3
		trigger = {

			exists = scope:super_company_reclamation_state_3
			scope:super_company_reclamation_state_3 = {
				owner = root
				super_company_reclamation_state_eligible = yes
			}
		}
		custom_tooltip = super_company_reclamation_preview_tt
		hidden_effect = {
			scope:super_company_reclamation_state_3 = {
				save_scope_as = super_company_reclamation_selected_state
			}
			trigger_event = { id = super_company_reclamation.3 }
		}
	}
	option = {
		name = super_company_reclamation.2.state_4
		trigger = {
			super_company_reclamation_country_eligible = yes
			exists = scope:super_company_reclamation_state_4
			scope:super_company_reclamation_state_4 = {
				owner = root
				super_company_reclamation_state_eligible = yes
			}
		}
		custom_tooltip = super_company_reclamation_preview_tt
		hidden_effect = {
			scope:super_company_reclamation_state_4 = {
				save_scope_as = super_company_reclamation_selected_state
			}
			trigger_event = { id = super_company_reclamation.3 }
		}
	}
	option = {
		name = super_company_reclamation.previous
		trigger = {
			super_company_reclamation_country_eligible = yes
			var:super_company_reclamation_offset >= 4
		}
		custom_tooltip = super_company_reclamation_page_tt
		hidden_effect = {
			change_variable = { name = super_company_reclamation_offset add = -4 }
			trigger_event = { id = super_company_reclamation.2 }
		}
	}
	option = {
		name = super_company_reclamation.next
		trigger = {
			super_company_reclamation_country_eligible = yes
			has_variable = super_company_reclamation_has_next
		}
		custom_tooltip = super_company_reclamation_page_tt
		hidden_effect = {
			change_variable = { name = super_company_reclamation_offset add = 4 }

		}
	}

		name = super_company_reclamation.cancel
		default_option = yes
		custom_tooltip = super_company_reclamation_cancel_tt
		hidden_effect = { super_company_reclamation_finish = yes }

				desc = super_company_reclamation.3.invalid.d

	option = {
		name = super_company_reclamation.3.a
		trigger = { super_company_reclamation_can_confirm = yes }
